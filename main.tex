% ---------- Theorem environments ----------
\newtheorem{theorem}{Theorem}[section]
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{definition}[theorem]{Definition}

% \autoref needs to be told these names; without this it calls every
% shared-counter environment "Theorem", and sections "section" (lowercase).
% (\providecommand first: hyperref already defines some of these.)
% Caveat: \autoref picks the name from the *counter*, so a shared-counter
% environment (definition/lemma below) autorefs as "Theorem". Either give it
% its own counter, write "Definition~\ref{...}" by hand, or use cleveref's
% \cref, which gets this right.
\providecommand{\theoremautorefname}{}\renewcommand{\theoremautorefname}{Theorem}
\providecommand{\lemmaautorefname}{}\renewcommand{\lemmaautorefname}{Lemma}
\providecommand{\definitionautorefname}{}\renewcommand{\definitionautorefname}{Definition}
\providecommand{\sectionautorefname}{}\renewcommand{\sectionautorefname}{Section}
\providecommand{\subsectionautorefname}{}\renewcommand{\subsectionautorefname}{Section}
